# ============================================
# ATC PLANNER - DOCKERFILE POUR HF SPACES
# ============================================

# 1. Image de base Python
FROM python:3.11-slim

# 2. Créer un utilisateur non-root (OBLIGATOIRE pour HF)
RUN useradd -m -u 1000 user
USER user
ENV PATH="/home/user/.local/bin:$PATH"

# 3. Dossier de travail
WORKDIR /app

# 4. Copier et installer les dépendances
COPY --chown=user requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# 5. Copier tout le code
COPY --chown=user . .

# 6. Port requis par HF Spaces (OBLIGATOIRE)
EXPOSE 7860

# 7. Variables d'environnement Streamlit
ENV STREAMLIT_SERVER_PORT=7860
ENV STREAMLIT_SERVER_ADDRESS=0.0.0.0
ENV STREAMLIT_SERVER_HEADLESS=true
ENV STREAMLIT_BROWSER_GATHER_USAGE_STATS=false

# 8. Commande de lancement
CMD ["streamlit", "run", "app.py", "--server.port=7860", "--server.address=0.0.0.0", "--server.headless=true"]
